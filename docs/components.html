Button

02

Primary appears once per view; Secondary is next-most-important; Ghost for anything tertiary. Loading state is imperative in vanilla JS (Fernwell.loading) — model it with component state in React instead.

<button class="fw-btn fw-btn-primary" type="button">Send invoice</button>
<button class="fw-btn fw-btn-secondary" type="button">Save draft</button>
<button class="fw-btn fw-btn-ghost" type="button">Cancel</button>
<button class="fw-btn fw-btn-danger" type="button">Delete</button>

<script>
// Loading state: disables the button, swaps in a spinner, restores on reset.
Fernwell.loading.setButtonLoading(btn, 'Sending…');
Fernwell.loading.resetButtonLoading(btn);
</script>
function SendInvoiceButton() {
  const [loading, setLoading] = useState(false);
  return (
    <button
      className={`fw-btn fw-btn-primary${loading ? ' fw-btn-loading' : ''}`}
      type="button"
      aria-busy={loading || undefined}
      disabled={loading}
      onClick={() => setLoading(true)}
    >
      {loading && <span className="fw-spinner fw-spinner-sm" aria-hidden="true" />}
      {loading ? 'Sending…' : 'Send invoice'}
    </button>
  );
}

Render

Icon

10

Renders a named SVG into a data-fw-icon element. Decorative by default (aria-hidden) — give the accessible name to a wrapping control instead, e.g. aria-label on an icon button. Sized in em and colored via inherited color, so it always matches the text or button around it. See the full set on the overview page.

<span class="fw-icon" data-fw-icon="star" aria-hidden="true"></span> Featured

<button class="fw-icon-btn" type="button" aria-label="Delete">
  <span data-fw-icon="trash" aria-hidden="true"></span>
</button>
function Icon({ name }) {
  const ref = useRef(null);
  useEffect(() => { window.Fernwell.icon.init(ref.current); }, [name]);
  return <span className="fw-icon" data-fw-icon={name} aria-hidden="true" ref={ref} />;
}

function DeleteButton({ onClick }) {
  return (
    <button className="fw-icon-btn" type="button" aria-label="Delete" onClick={onClick}>
      <Icon name="trash" />
    </button>
  );
}

Pagination

15

Static markup — wire href/onClick to your own paging logic.

<nav class="fw-pagination" aria-label="Pagination">
  <span class="fw-pag-info">Showing 1–20 of 84</span>
  <div class="fw-pag-controls">
    <a class="fw-btn-page is-disabled" aria-disabled="true">← Prev</a>
    <span class="fw-pag-label">Page 1 of 5</span>
    <a class="fw-btn-page" href="?page=2">Next →</a>
  </div>
</nav>
function Pagination({ page, pageCount, total, onChange }) {
  return (
    <nav className="fw-pagination" aria-label="Pagination">
      <span className="fw-pag-info">Showing {(page - 1) * 20 + 1}–{Math.min(page * 20, total)} of {total}</span>
      <div className="fw-pag-controls">
        <a
          className={`fw-btn-page${page <= 1 ? ' is-disabled' : ''}`}
          aria-disabled={page <= 1}
          onClick={() => page > 1 && onChange(page - 1)}
        >
          ← Prev
        </a>
        <span className="fw-pag-label">Page {page} of {pageCount}</span>
        <a className="fw-btn-page" onClick={() => onChange(page + 1)}>Next →</a>
      </div>
    </nav>
  );
}

                <input type="checkbox" data-fw-select-all /><span className="fw-box" />
                <span className="fw-sr-only">Select all rows</span>
              </label>
            </th>
            <th scope="col" aria-sort="none" data-fw-sort><button type="button" className="fw-table-sort">Client</button></th>
            <th scope="col" className="fw-table-num" aria-sort="none" data-fw-sort><button type="button" className="fw-table-sort">Amount</button></th>
            <th scope="col">Status</th>
          </tr>
        </thead>
        <tbody>
          {rows.map((row) => (
            <tr key={row.id}>
              <td className="fw-table-select">
                <label className="fw-checkbox">
                  <input type="checkbox" data-fw-select-row /><span className="fw-box" />
                  <span className="fw-sr-only">Select {row.client}</span>
                </label>
              </td>
              <td>{row.client}</td>
              <td className="fw-table-num" data-fw-sort-value={row.amount}>${row.amount.toFixed(2)}</td>
              <td><span className="fw-tag fw-tag-pill fw-tag-live">{row.status}</span></td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

function CopyButton() {
